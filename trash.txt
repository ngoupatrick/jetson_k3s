sudo nmcli connection modify "Wired connection 1" +ipv4.addresses 10.1.10.1/24
# Appliquer la modification
sudo nmcli connection up "Wired connection 1"
# check
ip addr show eth0

# Activer l'IP Forwarding (Routage) 
echo "net.ipv4.ip_forward=1" | sudo tee -a /etc/sysctl.conf
sudo sysctl -p

# On cible le trafic venant du réseau 10.1.10.0/24 sortant par l'IP internet
sudo iptables -t nat -A POSTROUTING -s 10.1.10.0/24 -o eth0 -j MASQUERADE
# 3. Autoriser le transfert de paquets
sudo iptables -A FORWARD -i eth0 -o eth0 -j ACCEPT

# Pour sauvegarder les règles au redémarrage
sudo apt install iptables-persistent
# rules files & folders: /etc/iptables/rules.v4 | /etc/iptables/rules.v6
sudo netfilter-persistent save
# run-parts: executing /usr/share/netfilter-persistent/plugins.d/15-ip4tables save
# run-parts: executing /usr/share/netfilter-persistent/plugins.d/25-ip6tables save

-- debut code
import cv2
import os
import sys

# Récupération de l'URL via les variables d'environnement du Pod
# Format attendu : rtmp://mediamtx-service:1935/host_stream
rtmp_url = os.getenv('RTMP_URL', 'rtmp://mediamtx-service:1935/host_stream')

# PIPELINE GSTREAMER OPTIMISÉ POUR JETSON NANO (JetPack 4.6.6)
# 1. rtmpsrc : Capture le flux RTMP
# 2. flvdemux/h264parse : Extrait le H264
# 3. nvv4l2decoder : Décodage matériel sur le GPU (NVDEC)
# 4. nvvidconv : Transfert de la mémoire GPU (NVMM) vers la mémoire système (Raw)
# 5. videoconvert : Conversion finale en format BGR pour OpenCV
gst_pipeline = (
    f"rtmpsrc location={rtmp_url} ! "
    "flvdemux ! h264parse ! nvv4l2decoder ! "
    "nvvidconv ! video/x-raw, format=BGRx ! "
    "videoconvert ! video/x-raw, format=BGR ! appsink drop=true"
)

def main():
    print(f"Initialisation du flux : {rtmp_url}")
    
    # Tentative d'ouverture avec le backend GStreamer
    cap = cv2.VideoCapture(gst_pipeline, cv2.CAP_GSTREAMER)

    if not cap.isOpened():
        print("Erreur : Impossible d'ouvrir le pipeline GStreamer.")
        print("Vérifiez la connectivité vers MediaMTX et la présence des drivers NVIDIA.")
        sys.exit(1)

    print("Connexion établie. Début du traitement GPU...")
    
    try:
        frame_count = 0
        while True:
            ret, frame = cap.read()
            if not ret:
                print("Flux interrompu.")
                break

            # --- LOGIQUE DE TRAITEMENT ---
            # Exemple : On affiche un log toutes les 300 frames
            frame_count += 1
            if frame_count % 300 == 0:
                print(f"Frames traitées : {frame_count} | Résolution : {frame.shape[1]}x{frame.shape[0]}")

    except KeyboardInterrupt:
        print("Arrêt par l'utilisateur.")
    finally:
        cap.release()
        print("Ressources libérées.")

if __name__ == "__main__":
    main()

=== Explain ===
1. Les chiffres clés:
    Throughput (Débit) : 2010.57 qps
    	C'est le nombre d'images traitées par seconde. Votre Jetson Nano est capable d'analyser environ 2010 images par seconde avec ce modèle (probablement MNIST, qui est très léger).
    Latency (Latence) : 0.40 ms (moyenne)
    	C'est le temps que met le GPU pour traiter une seule image. C'est extrêmement rapide (moins d'une demi-milliseconde).

2. Le cycle de données (Host ↔ GPU)
Le rapport détaille le temps passé à chaque étape :
    H2D (Host to Device) : 0.047 ms → Temps pour envoyer l'image de la RAM vers la mémoire GPU.
    GPU Compute Time : 0.315 ms → Le calcul pur de l'IA par les cœurs CUDA.
    D2H (Device to Host) : 0.044 ms → Temps pour récupérer le résultat (ex: "C'est un chiffre 5") du GPU vers le CPU.

3. L'avertissement important (La dernière ligne)
"Throughput may be bound by Enqueue Time rather than GPU Compute and the GPU may be under-utilized."
	Cela signifie que votre CPU est trop lent par rapport à la vitesse du GPU pour ce modèle précis.
	Le CPU passe plus de temps à "préparer et envoyer" les ordres (Enqueue Time) qu'à laisser le GPU travailler.
    
Conclusion : 
	Votre GPU ne travaille pas à 100% de sa capacité car il attend que le CPU lui donne du travail. C'est normal sur des modèles très simples comme MNIST sur une Jetson Nano.
	Cela prouve que votre installation CUDA/TensorRT sur JetPack 4.6.6 est parfaitement optimisée. Le GPU répond instantanément.

Comment cela fonctionne concrètement :
    L'Operator : Il surveille votre cluster K3s.
    Le MPIJob (YAML) : Vous lui envoyez un fichier de configuration qui dit : "Je veux 1 Launcher (le chef) et 2 Workers (s1 et s2) utilisant chacun 1 GPU".
    L'exécution : L'Operator crée automatiquement les Pods, configure les clés SSH entre eux (nécessaire pour MPI) et lance votre script d'entraînement via horovodrun.

# Image de base officielle NVIDIA pour JetPack 4.6.x
FROM nvcr.io/nvidia/l4t-ml:r32.7.1-py3

# Configuration environnement pour CUDA sur Jetson
ENV PATH=/usr/local/cuda/bin:${PATH}
ENV LD_LIBRARY_PATH=/usr/local/cuda/lib64:${LD_LIBRARY_PATH}
ENV HOROVOD_GPU_OPERATIONS=NCCL

# Installation des dépendances pour MPI et Horovod
RUN apt-get update && apt-get install -y --no-install-recommends \
    cmake \
    g++-7 \
    openmpi-bin \
    libopenmpi-dev \
    openssh-client \
    openssh-server \
    && rm -rf /var/lib/apt/lists/*

# Installation de Horovod (Prend du temps car compile sur Jetson)
# On cible PyTorch ici, adaptez si vous utilisez TensorFlow
RUN HOROVOD_WITH_PYTORCH=1 \
    pip3 install --no-cache-dir horovod[pytorch]

# Dossier de travail
WORKDIR /app
#COPY train.py /app/train.py

# Configuration SSH pour MPI (nécessaire pour le Training Operator)
RUN mkdir -p /var/run/sshd && \
    echo "PermitRootLogin yes" >> /etc/ssh/sshd_config && \
    echo "PasswordAuthentication no" >> /etc/ssh/sshd_config

import torch
import torchvision.models as models
import horovod.torch as hvd

# 1. Initialisation Horovod
hvd.init()
torch.cuda.set_device(0) # Toujours 0 sur Jetson

# 2. Chargement du modèle ResNet
# 'weights' permet de charger un modèle pré-entraîné (Transfer Learning)
model = models.resnet50(weights=models.ResNet50_Weights.DEFAULT)
model.cuda()

# 3. Ajuster le taux d'apprentissage (Learning Rate)
# On multiplie par le nombre de GPUs (hvd.size()) car le batch global est plus grand
lr = 0.001 * hvd.size()
optimizer = torch.optim.SGD(model.parameters(), lr=lr, momentum=0.9)

# 4. Ajouter le Optimizer Horovod
# Il va synchroniser les gradients entre master, s1 et s2
optimizer = hvd.DistributedOptimizer(
    optimizer, 
    named_parameters=model.named_parameters()
)

# 5. Diffuser les poids initiaux du master vers tous les workers
hvd.broadcast_parameters(model.state_dict(), root_rank=0)
