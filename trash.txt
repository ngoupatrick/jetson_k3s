gpu_b = cpu_b.to(device)

speedup = (end_cpu - start_cpu) / (end_gpu - start_gpu)

-- debut code
print("-- Import libs --")
import torch
import torch.nn as nn
import torch.optim as optim
from torchvision import datasets, transforms, models
import time

# Configuration
print("-- Init device --")
DEVICE = torch.device("cuda" if torch.cuda.is_available() else "cpu")
LOCAL_WEIGHTS = "/mnt/code/models/resnet50-0676ba61.pth"
EPOCHS = 10
BATCH_SIZE = 8  # Reduit pour eviter OOM sur Jetson avec ResNet50
DATA_DIR = '/mnt/code/images'
SAVE_PATH = '/mnt/code/car_model.pth'

# 1. Pretraitement
print("-- Pretraitement --")
transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.RandomHorizontalFlip(), # Augmentation de donnees
    transforms.ToTensor(),
    transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225])
])

# 2. Chargement des donnees
print("-- Chargement des donnees --")
dataset = datasets.ImageFolder(root=DATA_DIR, transform=transform)
train_loader = torch.utils.data.DataLoader(dataset, batch_size=BATCH_SIZE, shuffle=True)
print(f"Classes detectees : {dataset.class_to_idx}")
num_classes = len(dataset.classes)
print(f"Nombre de classes : {num_classes}")

# 3. Modele ResNet50
print("-- Chargement Modele ResNet50 --")
model = models.resnet50(pretrained=False)
model.load_state_dict(torch.load(LOCAL_WEIGHTS))
# Modifier la dernière couche pour les classes
model.fc = nn.Linear(model.fc.in_features, num_classes)
model = model.to(DEVICE)

criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=0.0001)

# 4. Boucle d'entrainement
print("-- Debut Boucle d'entrainement --")
print(f"Debut de l'entrainement sur {DEVICE}...")
model.train()

for epoch in range(EPOCHS):
    running_loss = 0.0
    start_time = time.time()
    
    for images, labels in train_loader:
        images, labels = images.to(DEVICE), labels.to(DEVICE)
        
        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item()

    epoch_loss = running_loss / len(train_loader)
    duration = time.time() - start_time
    print(f"Epoque [{epoch+1}/{EPOCHS}] - Loss: {epoch_loss:.4f} - Temps: {duration:.2f}s")

# 5. Sauvegarde
print(f"-- Sauvegarde {SAVE_PATH} --")
torch.save(model.state_dict(), SAVE_PATH)
print(f"Modele sauvegarde dans {SAVE_PATH}")

-- fin code

-- debut code
print(f"-- import libs --")
import torch
import torch.nn as nn
from torchvision import models, transforms
from PIL import Image
import os
import sys

print(f"-- Init Device --")
DEVICE = torch.device("cuda" if torch.cuda.is_available() else "cpu")
MODEL_PATH = "/mnt/code/car_model.pth"
# L'ordre doit correspondre à l'ordre alphabetique des dossiers
# {'empty': 0, 'plastic': 1, 'with-beer': 2, 'with-coffee': 3, 'with-juice': 4, 'with-milk': 5, 'with-tea': 6, 'with-water': 7, 'with-wine': 8}
CLASSES = ['empty', 'plastic', 'with-beer', 'with-coffee', 'with-juice', 'with-milk', 'with-tea', 'with-water', 'with-wine']

def predict_folder(folder_path):
    # 1. Configuration du modele
    print(f"-- 1. Configuration du modele --")
    model = models.resnet50()
    model.fc = nn.Linear(model.fc.in_features, len(CLASSES))
    if not os.path.exists(MODEL_PATH):
        print(f"Erreur : Modèle non trouve ici {MODEL_PATH}")
        return
    
    model.load_state_dict(torch.load(MODEL_PATH))
    model = model.to(DEVICE).eval()

    # 2. Transformations
    print(f"-- 2. Transformations --")
    transform = transforms.Compose([
        transforms.Resize((224, 224)),
        transforms.ToTensor(),
        transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225])
    ])

    # 3. Boucle sur les images
    print(f"Analyse des images dans : {folder_path}")
    valid_extensions = ('.jpg', '.jpeg', '.png', '.bmp')
    
    files = [f for f in os.listdir(folder_path) if f.lower().endswith(valid_extensions)]
    
    if not files:
        print("Aucune image trouvee dans le dossier.")
        return

    with torch.no_grad():
    	print(f"Boucle prediction des images dans : {folder_path}")
        for filename in files:
            img_path = os.path.join(folder_path, filename)
            try:
                img = Image.open(img_path).convert('RGB')
                img_tensor = transform(img).unsqueeze(0).to(DEVICE)
                
                output = model(img_tensor)
                _, pred = torch.max(output, 1)
                
                print(f"Image: {filename:25} | Resultat: {CLASSES[pred.item()]}")
            except Exception as e:
                print(f"Erreur sur {filename}: {e}")

if __name__ == "__main__":
    # Dossier par defaut si aucun argument n'est passee
    target_folder = sys.argv[1] if len(sys.argv) > 1 else "/mnt/code/test"
    predict_folder(target_folder)

=== Explain ===
1. Les chiffres clés:
    Throughput (Débit) : 2010.57 qps
    	C'est le nombre d'images traitées par seconde. Votre Jetson Nano est capable d'analyser environ 2010 images par seconde avec ce modèle (probablement MNIST, qui est très léger).
    Latency (Latence) : 0.40 ms (moyenne)
    	C'est le temps que met le GPU pour traiter une seule image. C'est extrêmement rapide (moins d'une demi-milliseconde).

2. Le cycle de données (Host ↔ GPU)
Le rapport détaille le temps passé à chaque étape :
    H2D (Host to Device) : 0.047 ms → Temps pour envoyer l'image de la RAM vers la mémoire GPU.
    GPU Compute Time : 0.315 ms → Le calcul pur de l'IA par les cœurs CUDA.
    D2H (Device to Host) : 0.044 ms → Temps pour récupérer le résultat (ex: "C'est un chiffre 5") du GPU vers le CPU.

3. L'avertissement important (La dernière ligne)
"Throughput may be bound by Enqueue Time rather than GPU Compute and the GPU may be under-utilized."
	Cela signifie que votre CPU est trop lent par rapport à la vitesse du GPU pour ce modèle précis.
	Le CPU passe plus de temps à "préparer et envoyer" les ordres (Enqueue Time) qu'à laisser le GPU travailler.
    
Conclusion : 
	Votre GPU ne travaille pas à 100% de sa capacité car il attend que le CPU lui donne du travail. C'est normal sur des modèles très simples comme MNIST sur une Jetson Nano.
	Cela prouve que votre installation CUDA/TensorRT sur JetPack 4.6.6 est parfaitement optimisée. Le GPU répond instantanément.

Comment cela fonctionne concrètement :
    L'Operator : Il surveille votre cluster K3s.
    Le MPIJob (YAML) : Vous lui envoyez un fichier de configuration qui dit : "Je veux 1 Launcher (le chef) et 2 Workers (s1 et s2) utilisant chacun 1 GPU".
    L'exécution : L'Operator crée automatiquement les Pods, configure les clés SSH entre eux (nécessaire pour MPI) et lance votre script d'entraînement via horovodrun.

import torch
import torchvision.models as models
import horovod.torch as hvd

# 1. Initialisation Horovod
hvd.init()
torch.cuda.set_device(0) # Toujours 0 sur Jetson

# 2. Chargement du modèle ResNet
# 'weights' permet de charger un modèle pré-entraîné (Transfer Learning)
model = models.resnet50(weights=models.ResNet50_Weights.DEFAULT)
model.cuda()

# 3. Ajuster le taux d'apprentissage (Learning Rate)
# On multiplie par le nombre de GPUs (hvd.size()) car le batch global est plus grand
lr = 0.001 * hvd.size()
optimizer = torch.optim.SGD(model.parameters(), lr=lr, momentum=0.9)

# 4. Ajouter le Optimizer Horovod
# Il va synchroniser les gradients entre master, s1 et s2
optimizer = hvd.DistributedOptimizer(
    optimizer, 
    named_parameters=model.named_parameters()
)

# 5. Diffuser les poids initiaux du master vers tous les workers
hvd.broadcast_parameters(model.state_dict(), root_rank=0)
